Trip Generation

viewof globalCalibRun = {
  const options = calib_runs_ojs;
  if (options.length === 0) {
    return Inputs.select([], {label: "Calibration Run:", disabled: true});
  }
  // Default to the run with the highest numeric ID (e.g. "C51" over "C50"),
  // not just the last entry in `options` -- that list comes from a plain
  // alphabetical sort (see list_available_runs in _validation_scripts.py),
  // which only happens to match numeric order today because both IDs have
  // the same digit count ("C9" would sort after "C50" alphabetically despite
  // being numerically earlier).
  const runNumber = id => {
    const m = /(\d+)/.exec(id);
    return m ? parseInt(m[1], 10) : -Infinity;
  };
  const latestRun = options.reduce((best, id) => runNumber(id) > runNumber(best) ? id : best, options[0]);
  const stored = window.localStorage.getItem("tdmcalib:selectedRun");
  const initial = options.includes(stored) ? stored : latestRun;
  const input = Inputs.select(options, {label: "Calibration Run:", value: initial});
  const selectEl = input.querySelector("select");
  input.classList.add("calib-run-bar");

  // Sidebar links for pages that don't have data for every run (currently
  // just report/cvm-update.qmd, which is only ever generated for the model
  // version's reference run -- not planned for every run the way the other
  // report pages are) get greyed out -- rather than the run dropdown itself
  // shrinking to that page's subset -- so the dropdown's options stay
  // identical everywhere on the site. The link stays clickable. If the
  // current page happens to be cvm-update.qmd itself, this also swaps its
  // own #cvm-content/#cvm-empty-message visibility -- both elements are
  // absent on every other page, so this is a no-op there.
  function updateSidebarAvailability(run) {
    const available = cvm_update_available_runs_ojs.includes(run);

    const link = document.querySelector('#quarto-sidebar a[href$="cvm-update.html"]');
    if (link) {
      const item = link.closest(".sidebar-item") ?? link;
      item.classList.toggle("sidebar-item-no-run-data", !available);
      link.title = available ? "" : `No commercial-vehicle factoring data for ${run}`;
    }

    const content = document.getElementById("cvm-content");
    const emptyMsg = document.getElementById("cvm-empty-message");
    if (content) content.style.display = available ? "" : "none";
    if (emptyMsg) emptyMsg.style.display = available ? "none" : "";
  }
  updateSidebarAvailability(initial);

  input.addEventListener("input", () => {
    window.localStorage.setItem("tdmcalib:selectedRun", input.value);
    updateSidebarAvailability(input.value);
  });

  // Quarto's own OJS runtime places this cell's returned node into its
  // normal in-page output slot right after this function returns -- moving
  // it into the header/sidebar synchronously, before that happens, would
  // just get undone a moment later. Deferring to the next tick runs this
  // after that placement, so the move actually sticks.
  setTimeout(() => {
    const navbarBrand = document.querySelector("#quarto-header .navbar-brand-container");
    if (navbarBrand) {
      input.classList.remove("calib-run-bar");
      input.classList.add("calib-run-navbar");
      navbarBrand.appendChild(input);
      return;
    }

    const sidebar = document.querySelector("#quarto-sidebar");
    const menu = sidebar?.querySelector(".sidebar-menu-container");
    if (!sidebar || !menu) return; // no navbar brand container or docked sidebar -- stays inline as a .calib-run-bar
    // Inputs.select renders the label text and <select> on the same line by
    // default -- force them onto separate lines to match the sidebar's
    // stacked item style. Insert relative to the select's actual parent
    // (not assumed to be the <label>, which varies by Inputs version) so
    // this doesn't throw if they aren't nested.
    selectEl?.parentNode.insertBefore(document.createElement("br"), selectEl);
    input.classList.remove("calib-run-bar");
    input.classList.add("calib-run-sidebar");
    sidebar.insertBefore(input, menu);
  }, 0);

  return input;
}

Trip Production Rates

The sources of the data include the most recent Utah Household Travel Survey (HTS) from 2023, the HTS from 2012, and the base year model (BY 2023 Model).

function tripDumbbellChart(county) {
  const htsSource = "HTS " + htsYearSelect;

  // Reactive data -- always keep the selected HTS reference year (calib_run
  // == null), but only the selected calibration run's Model rows
  const filtered = tripRates
    .filter(d => d.County === county)
    .filter(d => d.calib_run == null || d.calib_run === globalCalibRun)
    .map(d => ({ ...d, "Data Source": String(d["Data Source"]).trim() }));

  // "All" (the county's total across every purpose) runs 4-8x larger than
  // any individual purpose -- including it would stretch the shared x-axis
  // so far that the individual purposes collapse together near zero. Shown
  // in the totals table below the grid instead (see tripTotalsRow).
  const purposes = purposeOrder.filter(p => p !== "All" && filtered.some(d => d.Purpose === p));

  const rows = purposes.map(p => {
    const hts = filtered.find(d => d.Purpose === p && d["Data Source"] === htsSource);
    const mod = filtered.find(d => d.Purpose === p && d["Data Source"] === "Model");
    return { Purpose: p, HTS: hts?.[metricSelect], Model: mod?.[metricSelect] };
  }).filter(d => d.HTS != null && d.Model != null);

  const xMax = d3.max(rows, d => Math.max(d.HTS, d.Model)) || 1;
  // Chart sits in one cell of the 2x3 grid (trip-dumbbell-grid below) --
  // account for the cell being a third of the container width above the
  // grid's breakpoint, and full width once the grid collapses to one
  // column, so the plotted width matches what CSS actually gives this chart.
  const chartWidth = width < tripGridBreakpoint ? width : (width - tripGridGap * 2) / 3;

  const chart = Plot.plot({
    width: chartWidth,
    height: 260,
    marginLeft: 70,
    marginRight: 20,
    marginTop: 10,
    y: {domain: purposes, label: null},
    x: {label: metricSelect, domain: [0, xMax], grid: true, nice: true},
    marks: [
      Plot.link(rows, {
        x1: "HTS", x2: "Model", y: "Purpose",
        stroke: themeLinkGrey, strokeWidth: 3
      }),
      Plot.dot(rows, {
        x: "HTS", y: "Purpose", fill: themeTarget, r: 7,
        title: d => `${d.Purpose}\n${htsSource}: ${d3.format(",.2f")(d.HTS)}`
      }),
      Plot.dot(rows, {
        x: "Model", y: "Purpose", fill: themeModel, r: 7,
        title: d => `${d.Purpose}\nModel (${globalCalibRun}): ${d3.format(",.2f")(d.Model)}`
      })
    ]
  });

  return html`<div>
    <div style="font-weight:600; text-align:center; margin-bottom:2px;">${county}</div>
    ${chart}
  </div>`;
}

// One row of "All"-purpose totals per county, left out of the dumbbell
// panels above (see tripDumbbellChart) since their magnitude would dominate
// the shared per-panel x-axis.
function tripTotalsRow(county) {
  const htsSource = "HTS " + htsYearSelect;
  const filtered = tripRates
    .filter(d => d.County === county && d.Purpose === "All")
    .filter(d => d.calib_run == null || d.calib_run === globalCalibRun)
    .map(d => ({ ...d, "Data Source": String(d["Data Source"]).trim() }));
  const hts = filtered.find(d => d["Data Source"] === htsSource);
  const mod = filtered.find(d => d["Data Source"] === "Model");
  return { County: county, HTS: hts?.[metricSelect], Model: mod?.[metricSelect] };
}

Commercial Vehicle Trip Generation

Unlike person trips, there’s no household-travel-survey equivalent for commercial vehicles, so there’s no independent target to validate this against (see the Commercial Vehicle Model Update page’s “About the LOCUS Data” note, and the Trip Distribution page’s trip-length-frequency note). The chart below is a plain model summary – daily commercial-vehicle trips produced, by county and vehicle type.
