Trip Generation

viewof globalCalibRun = {
  const options = calib_runs_ojs;
  if (options.length === 0) {
    return Inputs.select([], {label: "Calibration Run:", disabled: true});
  }
  // Default to the run with the highest numeric ID (e.g. "C51" over "C50"),
  // not just the last entry in `options` -- that list comes from a plain
  // alphabetical sort (see list_available_runs in _validation_scripts.py),
  // which only happens to match numeric order today because both IDs have
  // the same digit count ("C9" would sort after "C50" alphabetically despite
  // being numerically earlier).
  const runNumber = id => {
    const m = /(\d+)/.exec(id);
    return m ? parseInt(m[1], 10) : -Infinity;
  };
  const latestRun = options.reduce((best, id) => runNumber(id) > runNumber(best) ? id : best, options[0]);
  const stored = window.localStorage.getItem("tdmcalib:selectedRun");
  const initial = options.includes(stored) ? stored : latestRun;
  const input = Inputs.select(options, {label: "Calibration Run:", value: initial});
  // Inputs.select renders the label text and <select> on the same line by
  // default -- force them onto separate lines. Insert relative to the
  // select's actual parent (not assumed to be the <label>, which varies by
  // Inputs version) so this doesn't throw if they aren't nested.
  const selectEl = input.querySelector("select");
  selectEl?.parentNode.insertBefore(document.createElement("br"), selectEl);
  input.classList.add("calib-run-bar");
  input.addEventListener("input", () => {
    window.localStorage.setItem("tdmcalib:selectedRun", input.value);
  });

  // Quarto's own OJS runtime places this cell's returned node into its
  // normal in-page output slot right after this function returns -- moving
  // it into the sidebar synchronously, before that happens, would just get
  // undone a moment later. Deferring to the next tick runs this after that
  // placement, so the move actually sticks.
  setTimeout(() => {
    const sidebar = document.querySelector("#quarto-sidebar");
    const menu = sidebar?.querySelector(".sidebar-menu-container");
    if (!sidebar || !menu) return; // no docked sidebar -- stays inline as a .calib-run-bar
    input.classList.remove("calib-run-bar");
    input.classList.add("calib-run-sidebar");
    sidebar.insertBefore(input, menu);
  }, 0);

  return input;
}

Trip Production Rates

The sources of the data include the most recent Utah Household Travel Survey (HTS) from 2023, the HTS from 2012, and the base year model (BY 2023 Model).

areas = ["Region","Box Elder","Weber","Davis","Salt Lake","Utah"]
areasAvailable = areas.filter(a => tripRates.some(d => d.County === a))

metricMap = new Map([
  ["Trips per Household", "Trips per Household"],
  ["Trips per Person", "Trips per Person"]
])

viewof countySelect = Inputs.select(areasAvailable, {
  value: "Region",
  label: "Area"
})

viewof metricSelect = Inputs.select(metricMap, {
  value: "Trips per Household",
  label: "Metric:"

import {GroupedBarChart} from "@d3/grouped-bar-chart"
import {Swatches} from "@d3/color-legend"

{
  const container = html`
    <div style="display:flex; flex-direction:column; gap:10px;"></div>
  `;

  // Controls
  const controls = html`
    <div style="display:flex; gap:16px; align-items:center; flex-wrap:wrap;"></div>
  `;
  controls.append(viewof countySelect);
  controls.append(viewof metricSelect);
  container.append(controls);

  // Reactive data -- always keep HTS reference rows (calib_run == null),
  // but only the selected calibration run's Model rows
  const filtered_data = tripRates
    .filter(d => d.County === countySelect)
    .filter(d => d.calib_run == null || d.calib_run === globalCalibRun)
    .map(d => ({ ...d, "Data Source": String(d["Data Source"]).trim() }));

  const purposeOrder = ["All","HBW","HBShp","HBOth","HBSch","NHBW","NHBNW","IX"];
  const purposes = purposeOrder.filter(p =>
    filtered_data.some(d => d.Purpose === p)
  );

  // Title
  container.append(html`
    <div style="font-size:18px; font-weight:600; margin-top:8px;">
      ${metricSelect} — ${countySelect} — ${globalCalibRun}
    </div>
  `);

  // Chart
  const chart = GroupedBarChart(filtered_data, {
    x: d => d.Purpose,
    y: d => d[metricSelect],
    z: d => d["Data Source"],
    xDomain: purposes,
    zDomain: ["HTS 2012", "HTS 2023", "Model"],
    colors: ["lightgray", themeTarget, themeModel],
    width: 900
  });

  chart.querySelectorAll("text").forEach(t => (t.style.fontSize = "18px"));
  container.append(chart);

  // Legend
  const color = d3.scaleOrdinal(
    ["HTS 2012", "HTS 2023", "BY 2023 Model"],
    ["lightgray", themeTarget, themeModel]
  );
  container.append(Swatches(color, { label: "Data Source" }));

  return container;
}
